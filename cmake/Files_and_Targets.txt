#The FLEUR executable. There is only ever one of it in a build tree: the MPI and
#the serial version differ in the compile definitions and the linked solver
#libraries, not in the target. The binary is named 'fleur_MPI' instead of
#'fleur' for a parallel build (see RUNTIME_OUTPUT_NAME below), because the test
#system and the packaging recipes identify the variant by that file name.
add_executable(fleur ${fleur_SRC} ${c_filesFleur})
target_compile_definitions(fleur PUBLIC ${FLEUR_DEFINITIONS} ${FLEUR_PARALLEL_DEFINITIONS})
target_compile_options(fleur PUBLIC $<$<COMPILE_LANGUAGE:Fortran>:${FLEUR_PRECISION_OPTION}>)
target_compile_options(fleur PRIVATE $<$<COMPILE_LANGUAGE:Fortran>:${FLEUR_COMPILE_OPTIONS}>)
target_compile_options(fleur PRIVATE -I${CMAKE_BINARY_DIR}/modules/fleur)
set_target_properties(fleur PROPERTIES Fortran_MODULE_DIRECTORY ${CMAKE_BINARY_DIR}/modules/fleur)
target_link_libraries(fleur fleurinput juDFT)

if (FLEUR_COMPILE_EDSOLVER)
   target_link_libraries(fleur EDsolver)
endif()
if (FLEUR_COMPILE_MAGMA)
   target_link_libraries(fleur magma)
endif()

if (FLEUR_USE_MPI)
   set_target_properties(fleur PROPERTIES RUNTIME_OUTPUT_NAME fleur_MPI)
   #Parallel eigensolvers are only available in the MPI build
   if (FLEUR_COMPILE_CHASE)
      target_link_libraries(fleur chase_f)
      set_target_properties(fleur PROPERTIES LINKER_LANGUAGE Fortran)
   endif()
   if (FLEUR_COMPILE_ELSI)
      target_link_libraries(fleur elsi)
      set_target_properties(fleur PROPERTIES LINKER_LANGUAGE Fortran)
   endif()
   if (FLEUR_USE_INTERNAL_ELPA)
     add_dependencies(fleur elpa)
     target_link_libraries(fleur elpa)
     target_link_libraries(fleur "-lstdc++")
     add_dependencies(elpa scalapack)
   endif()
   IF (FLEUR_COMPILE_SCALAPACK)
      target_link_libraries(fleur scalapack)
      if (FLEUR_COMPILE_ELSI)
      cmake_policy(SET CMP0079 NEW)
      target_link_libraries(elpa PRIVATE scalapack)
      target_link_libraries(chase PRIVATE scalapack)
      endif()
   endif()
endif ()

if (FLEUR_USE_LIBXC AND FLEUR_LIBXC_TARGETS)

endif()
