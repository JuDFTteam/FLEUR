# CMakeLists.txt for the FLEURiste TUI / CLI application
#
# This target creates a Python virtual environment in the build directory,
# installs the fleuriste package (with its dependencies) into it, and
# generates a thin shell wrapper "fleuriste" that activates the venv and
# forwards all arguments to the installed entry-point.

set(FLEURISTE_SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(FLEURISTE_VENV    "${CMAKE_BINARY_DIR}/fleuriste-venv")
set(FLEURISTE_STAMP   "${CMAKE_CURRENT_BINARY_DIR}/.fleuriste_installed_stamp")
set(FLEURISTE_SCRIPT  "${CMAKE_BINARY_DIR}/fleuriste")

# Collect the Python source files so that a rebuild is triggered when
# any of them change.
file(GLOB FLEURISTE_PY_SOURCES
     "${FLEURISTE_SRC_DIR}/*.py"
     "${FLEURISTE_SRC_DIR}/pyjob/*.py"
)

# Generate the wrapper script at configure time
set(FLEURISTE_SCRIPT_TEMPLATE "${CMAKE_CURRENT_BINARY_DIR}/fleuriste.sh.in")
file(WRITE "${FLEURISTE_SCRIPT_TEMPLATE}"
"#!/usr/bin/env bash
# Auto-generated by CMake - do not edit.
# Thin wrapper that runs the FLEURiste CLI inside its virtual environment.

FLEURISTE_VENV=\"${FLEURISTE_VENV}\"
export FLEUR_BUILDDIR=\"${CMAKE_BINARY_DIR}\"
export DYLD_LIBRARY_PATH=\"${CMAKE_BINARY_DIR}/src/tools/inpgen3:\\\${DYLD_LIBRARY_PATH:-}\"
export LD_LIBRARY_PATH=\"${CMAKE_BINARY_DIR}/src/tools/inpgen3:\\\${LD_LIBRARY_PATH:-}\"

if [ ! -d \"\$FLEURISTE_VENV\" ]; then
    echo \"Error: FLEURiste virtual environment not found at \$FLEURISTE_VENV\" >&2
    echo \"Please rebuild the 'fleuriste' target.\" >&2
    exit 1
fi

exec \"\$FLEURISTE_VENV/bin/fleuriste\" \"\$@\"
")

# Step 1+2: create venv & pip-install the package into it.
# The stamp file avoids re-running pip on every build.
add_custom_command(
    OUTPUT  "${FLEURISTE_STAMP}" "${FLEURISTE_SCRIPT}"
    # Create the virtual environment (idempotent)
    COMMAND "${FLEUR_PYTHON}" -m venv "${FLEURISTE_VENV}"
    # Upgrade pip inside the venv
    COMMAND "${FLEURISTE_VENV}/bin/pip" install --upgrade pip -q
    # Install setuptools (required for editable installs)
    COMMAND "${FLEURISTE_VENV}/bin/pip" install setuptools wheel -q
    # Install fleuriste with jupyter extras into the venv
    COMMAND "${FLEURISTE_VENV}/bin/pip" install "${FLEURISTE_SRC_DIR}[jupyter]" -q
    # Install ipykernel and register this venv as a Jupyter kernel
    COMMAND "${FLEURISTE_VENV}/bin/pip" install ipykernel -q
    COMMAND "${FLEURISTE_VENV}/bin/python" -m ipykernel install --user --name fleuriste --display-name "Python FLEURiste"
    # Copy the wrapper script
    COMMAND "${CMAKE_COMMAND}" -E copy "${FLEURISTE_SCRIPT_TEMPLATE}" "${FLEURISTE_SCRIPT}"
    # Make script executable
    COMMAND chmod +x "${FLEURISTE_SCRIPT}"
    # Touch the stamp
    COMMAND "${CMAKE_COMMAND}" -E touch "${FLEURISTE_STAMP}"
    DEPENDS ${FLEURISTE_PY_SOURCES}
            "${FLEURISTE_SRC_DIR}/pyproject.toml"
    WORKING_DIRECTORY "${FLEURISTE_SRC_DIR}"
    COMMENT "Creating FLEURiste virtual environment and installing dependencies"
)

# Step 3: Create the target that builds everything
add_custom_target(fleuriste ALL
    DEPENDS "${FLEURISTE_STAMP}" "${FLEURISTE_SCRIPT}"
    COMMENT "FLEURiste CLI ready: ${FLEURISTE_SCRIPT}"
)
