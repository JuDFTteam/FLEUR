# CMakeLists.txt for the FLEURiste TUI / CLI application
#
# This target creates a Python virtual environment in the build directory,
# installs the fleuriste package (with its dependencies) into it, and
# generates a thin shell wrapper "fleuriste" that activates the venv and
# forwards all arguments to the installed entry-point.

# FLEURiste requires Python > 3.10 (i.e. 3.11+).
set(FLEUR_USE_FLEURISTE FALSE)
set(FLEUR_FLEURISTE_PYTHON_VERSION "not found")

if (FLEUR_PYTHON)
    execute_process(
        COMMAND "${FLEUR_PYTHON}" -c "import sys; print(f'{sys.version_info.major}.{sys.version_info.minor}')"
        RESULT_VARIABLE _fleuriste_python_result
        OUTPUT_VARIABLE _fleuriste_python_version
        ERROR_QUIET
        OUTPUT_STRIP_TRAILING_WHITESPACE
    )
    if (_fleuriste_python_result EQUAL 0)
        set(FLEUR_FLEURISTE_PYTHON_VERSION "${_fleuriste_python_version}")
        if (FLEUR_FLEURISTE_PYTHON_VERSION VERSION_GREATER "3.10")
            set(FLEUR_USE_FLEURISTE TRUE)
        endif()
    endif()
endif()

set(FLEUR_USE_FLEURISTE "${FLEUR_USE_FLEURISTE}" CACHE INTERNAL "Whether FLEURiste targets are enabled" FORCE)
set(FLEUR_FLEURISTE_PYTHON_VERSION "${FLEUR_FLEURISTE_PYTHON_VERSION}" CACHE INTERNAL "Python version used for FLEURiste checks" FORCE)

if (NOT FLEUR_USE_FLEURISTE)
    message(WARNING "FLEURiste disabled: Python > 3.10 required, detected '${FLEUR_FLEURISTE_PYTHON_VERSION}' from '${FLEUR_PYTHON}'")
    return()
endif()

set(FLEURISTE_SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(FLEURISTE_VENV    "${CMAKE_BINARY_DIR}/fleuriste-venv")
set(FLEURISTE_STAMP   "${CMAKE_CURRENT_BINARY_DIR}/.fleuriste_installed_stamp")
set(FLEURISTE_SCRIPT  "${CMAKE_BINARY_DIR}/FLEURiste")
set(FLEURISTE_STAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fleuriste-stage/fleuriste")
set(FLEURISTE_WHEEL_DIR "${CMAKE_CURRENT_BINARY_DIR}/fleuriste-wheel")
set(FLEURISTE_INSTALL_WHEEL_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/install_fleuriste_wheel.py")
set(FLEURISTE_INPGEN_WRAPPER "${FLEURISTE_SRC_DIR}/../FleurInpgen.py")

# Collect the Python source files so that a rebuild is triggered when
# any of them change.
file(GLOB FLEURISTE_PY_SOURCES
     "${FLEURISTE_SRC_DIR}/*.py"
     "${FLEURISTE_SRC_DIR}/pyjob/*.py"
)

# Generate the wrapper script at configure time
set(FLEURISTE_SCRIPT_TEMPLATE "${CMAKE_CURRENT_BINARY_DIR}/fleuriste.sh.in")
file(WRITE "${FLEURISTE_SCRIPT_TEMPLATE}"
"#!/usr/bin/env bash
# Auto-generated by CMake - do not edit.
# Thin wrapper that runs the FLEURiste CLI inside its virtual environment.

FLEURISTE_VENV=\"${FLEURISTE_VENV}\"
export FLEUR_BUILDDIR=\"${CMAKE_BINARY_DIR}\"
export DYLD_LIBRARY_PATH=\"${CMAKE_BINARY_DIR}/src/tools/inpgen3:\${DYLD_LIBRARY_PATH:-}\"
export LD_LIBRARY_PATH=\"${CMAKE_BINARY_DIR}/src/tools/inpgen3:\${LD_LIBRARY_PATH:-}\"

if [ ! -d \"\$FLEURISTE_VENV\" ]; then
    echo \"Error: FLEURiste virtual environment not found at \$FLEURISTE_VENV\" >&2
    echo \"Please rebuild the 'fleuriste' target.\" >&2
    exit 1
fi

exec \"\$FLEURISTE_VENV/bin/fleuriste\" \"\$@\"
")

# Helper script: install newest wheel from a wheel directory.
file(WRITE "${FLEURISTE_INSTALL_WHEEL_SCRIPT}"
"#!/usr/bin/env python3
import pathlib
import subprocess
import sys

wheel_dir = pathlib.Path(sys.argv[1])
pip_exe = sys.argv[2]
wheels = sorted(wheel_dir.glob('fleuriste-*.whl'))
if not wheels:
    raise SystemExit(f'No fleuriste wheel found in {wheel_dir}')

subprocess.check_call([pip_exe, 'install', '--force-reinstall', str(wheels[-1]), '-q'])
")

# Step 1+2: create venv and install package via wheel built from a staged copy.
# This avoids setuptools metadata writes (e.g., SOURCES.txt) in the source tree.
# The stamp file avoids re-running pip on every build.
add_custom_command(
    OUTPUT  "${FLEURISTE_STAMP}" "${FLEURISTE_SCRIPT}"
    # Create the virtual environment (idempotent)
    COMMAND "${FLEUR_PYTHON}" -m venv "${FLEURISTE_VENV}"
    # Upgrade pip inside the venv
    COMMAND "${FLEURISTE_VENV}/bin/pip" install --upgrade pip -q
    # Install packaging tools
    COMMAND "${FLEURISTE_VENV}/bin/pip" install setuptools wheel build -q
    # Create fresh staged source copy in build tree
    COMMAND "${CMAKE_COMMAND}" -E rm -rf "${FLEURISTE_STAGE_DIR}" "${FLEURISTE_WHEEL_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${FLEURISTE_SRC_DIR}" "${FLEURISTE_STAGE_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${FLEURISTE_INPGEN_WRAPPER}" "${CMAKE_CURRENT_BINARY_DIR}/fleuriste-stage/FleurInpgen.py"
        # Remove generated artifacts from staged copy to avoid recursive package contents
        COMMAND "${CMAKE_COMMAND}" -E rm -rf
            "${FLEURISTE_STAGE_DIR}/build"
            "${FLEURISTE_STAGE_DIR}/fleuriste.egg-info"
            "${FLEURISTE_STAGE_DIR}/__pycache__"
            "${FLEURISTE_STAGE_DIR}/pyjob/__pycache__"
    # Build wheel from staged source (never from source dir)
    COMMAND "${FLEURISTE_VENV}/bin/python" -m build --wheel --outdir "${FLEURISTE_WHEEL_DIR}" "${FLEURISTE_STAGE_DIR}"
    # Install the built wheel (pick newest wheel file in wheel dir)
        COMMAND "${FLEURISTE_VENV}/bin/python" "${FLEURISTE_INSTALL_WHEEL_SCRIPT}" "${FLEURISTE_WHEEL_DIR}" "${FLEURISTE_VENV}/bin/pip"
    # Install jupyter extra dependencies explicitly
    COMMAND "${FLEURISTE_VENV}/bin/pip" install -q "ipywidgets>=8.0" "jupyterlab-widgets>=3.0" "widgetsnbextension>=4.0" "matplotlib>=3.5" "jupyterlab>=3.0"
    # Install ipykernel and register this venv as a Jupyter kernel
    COMMAND "${FLEURISTE_VENV}/bin/pip" install ipykernel -q
    COMMAND "${FLEURISTE_VENV}/bin/python" -m ipykernel install --user --name fleuriste --display-name "Python FLEURiste"
    # Copy the wrapper script
    COMMAND "${CMAKE_COMMAND}" -E copy "${FLEURISTE_SCRIPT_TEMPLATE}" "${FLEURISTE_SCRIPT}"
    # Make script executable
    COMMAND chmod +x "${FLEURISTE_SCRIPT}"
    # Touch the stamp
    COMMAND "${CMAKE_COMMAND}" -E touch "${FLEURISTE_STAMP}"
    DEPENDS ${FLEURISTE_PY_SOURCES}
            "${FLEURISTE_INPGEN_WRAPPER}"
            "${FLEURISTE_SRC_DIR}/pyproject.toml"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    COMMENT "Creating FLEURiste virtual environment and installing dependencies"
)

# Step 3: Create the target that builds everything
add_custom_target(fleuriste ALL
    DEPENDS "${FLEURISTE_STAMP}" "${FLEURISTE_SCRIPT}"
    COMMENT "FLEURiste CLI ready: ${FLEURISTE_SCRIPT}"
)
